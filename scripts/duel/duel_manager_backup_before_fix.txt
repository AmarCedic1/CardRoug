extends Node
class_name DuelManager

var zones = {}          
var player_zones = {}   
var enemy_zones = {}    
var player_deck: Array = []
var player_hand: Array = []

func _ready():
    print("DuelManager ready")

    await get_tree().process_frame

    var db = get_node("../CardDatabase")
    print("GOAT cards:", db.cards.size())

    build_deck()

    for i in range(5):
        var card = draw_card()
        if card != null:
            add_card_to_hand(card)

    print("Starthand:", player_hand.size())



func build_deck():
    var db = get_node("../CardDatabase")

    var all_ids = db.cards.keys()  # api_id liste
    all_ids.shuffle()

    player_deck = []  # komplett neu setzen

    for i in range(40):
        player_deck.append(all_ids[i])

    print("Deck gebaut:", player_deck.size(), "Karten")



func register_zone(zone_name: String, zone_node: Node, is_player: bool):
    zones[zone_name] = zone_node

    if is_player:
        player_zones[zone_name] = zone_node
    else:
        enemy_zones[zone_name] = zone_node

    print("Zone registriert:", zone_name)

func get_zone(name: String) -> Node:
    return zones.get(name)
func draw_card():
    if player_deck.size() == 0:
        return null

    var api_id = player_deck.pop_back()

    var db = get_node("../CardDatabase")
    var card = db.get_card_by_id(api_id)

    return card


func add_card_to_hand(card_data: Dictionary):
    var ui_scene = load("res://ui/card_ui.tscn")
    var ui = ui_scene.instantiate()
    ui.set_card(card_data)

    var hand_zone = zones.get("PlayerHand")
    hand_zone.add_child(ui)

    var i = hand_zone.get_child_count() - 1
    ui.position = Vector2(20 + i * 150, 0)

    player_hand.append(card_data)

func normal_summon(card_ui):
    var zone = zones.get("PlayerMonster1")
    card_ui.get_parent().remove_child(card_ui)
    zone.add_child(card_ui)
    card_ui.position = Vector2.ZERO
    card_ui.in_attack_position = true
    card_ui.update_stats()

func set_summon(card_ui):
    var zone = zones.get("PlayerMonster1")
    card_ui.get_parent().remove_child(card_ui)
    zone.add_child(card_ui)
    card_ui.position = Vector2.ZERO
    card_ui.in_attack_position = false
    card_ui.update_stats()

func activate_spell(card_ui):
    var zone = zones.get("PlayerSpell1")
    card_ui.get_parent().remove_child(card_ui)
    zone.add_child(card_ui)
    card_ui.position = Vector2.ZERO

func set_spell(card_ui):
    var zone = zones.get("PlayerSpell1")
    card_ui.get_parent().remove_child(card_ui)
    zone.add_child(card_ui)
    card_ui.position = Vector2.ZERO

func on_card_clicked(card_ui):
    var menu = get_tree().root.get_node("main/ActionMenu")

    # Spell/Trap
    if card_ui.card_data["card_type"] == "spell" or card_ui.card_data["card_type"] == "trap":
        menu.get_node("NormalButton").visible = false
        menu.get_node("SetButton").visible = false
        menu.get_node("ActivateButton").visible = true
        menu.get_node("SetSpellButton").visible = true

    # Monster
    else:
        menu.get_node("NormalButton").visible = true
        menu.get_node("SetButton").visible = true
        menu.get_node("ActivateButton").visible = false
        menu.get_node("SetSpellButton").visible = false

    menu.open(card_ui, card_ui.get_global_position() + Vector2(80, 0))
